CREATE TABLE mascotas (
	id_mascota INT PRIMARY KEY AUTO_INCREMENT,
	nombre VARCHAR(50) NOT NULL,
	especie VARCHAR(30) NOT NULL,
	edad INT
);

CREATE TABLE veterinarios (
	id_veterinario INT PRIMARY KEY,
	nombre VARCHAR(50) NOT NULL,
	especialidad VARCHAR(50),
	salario DECIMAL(8,2)
);

CREATE TABLE consultas (
	id_consulta INT,
	fecha DATE,
	coste DECIMAL(6,2),
	id_mascota INT,
	id_veterinario INT,
	PRIMARY KEY (id_consulta, id_mascota, id_veterinario),
	CONSTRAINT con_idm_fk FOREIGN KEY (id_mascota)
		REFERENCES mascotas(id_mascota),
	CONSTRAINT con_idv_fk FOREIGN KEY (id_veterinario)
		REFERENCES veterinarios(id_veterinario)
);
